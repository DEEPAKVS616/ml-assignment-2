import streamlit as st
import pandas as pd
import numpy as np
import joblib
import matplotlib.pyplot as plt
from sklearn.metrics import (
    accuracy_score, roc_auc_score, precision_score, 
    recall_score, f1_score, matthews_corrcoef, 
    confusion_matrix, classification_report
)

st.set_page_config(page_title="ML Assignment 2 - Dashboard", layout="wide")

st.title("M.Tech AIML: Multi-Model Classification & Evaluation Dashboard")
st.markdown("**Course:** Machine Learning | **Assignment 2** | **Student Evaluation App**")

# Sidebar
st.sidebar.header("Configuration Panel")
model_name = st.sidebar.selectbox(
    "Select Classification Model",
    ["Logistic Regression", "Decision Tree", "KNN", "Naive Bayes", "Support Vector Machine"]
)

st.sidebar.markdown("---")
st.sidebar.subheader("Upload Test Dataset")
uploaded_file = st.sidebar.file_uploader("Upload test_data.csv", type=["csv"])

model_files = {
    'Logistic Regression': 'model/logistic_regression.pkl',
    'Decision Tree': 'model/decision_tree.pkl',
    'KNN': 'model/knn.pkl',
    'Naive Bayes': 'model/naive_bayes.pkl',
    'Support Vector Machine': 'model/support_vector_machine.pkl'
}

@st.cache_resource
def load_artifacts(path):
    return joblib.load(path)

try:
    scaler = load_artifacts('model/scaler.pkl')
except Exception:
    scaler = None

if uploaded_file is not None:
    test_df = pd.read_csv(uploaded_file)
    st.success("Test dataset loaded successfully!")
    
    if 'target' not in test_df.columns:
        st.error("Uploaded CSV must contain the 'target' column.")
    else:
        X_test = test_df.drop(columns=['target'])
        y_test = test_df['target']
        
        # Load model
        model = load_artifacts(model_files[model_name])
        
        # Preprocess if required
        if model_name in ['Logistic Regression', 'KNN', 'Support Vector Machine'] and scaler is not None:
            X_test_processed = scaler.transform(X_test)
        else:
            X_test_processed = X_test
            
        # Predict
        y_pred = model.predict(X_test_processed)
        try:
            y_prob = model.predict_proba(X_test_processed)[:, 1]
        except Exception:
            y_prob = np.zeros_like(y_pred, dtype=float)
            
        # Metrics
        acc = accuracy_score(y_test, y_pred)
        auc_val = roc_auc_score(y_test, y_prob) if len(np.unique(y_test)) > 1 else 0.0
        prec = precision_score(y_test, y_pred, zero_division=0)
        rec = recall_score(y_test, y_pred, zero_division=0)
        f1 = f1_score(y_test, y_pred, zero_division=0)
        mcc = matthews_corrcoef(y_test, y_pred)
        
        st.subheader(f"Performance Metrics: {model_name}")
        c1, c2, c3, c4, c5, c6 = st.columns(6)
        c1.metric("Accuracy", f"{acc:.4f}")
        c2.metric("AUC Score", f"{auc_val:.4f}")
        c3.metric("Precision", f"{prec:.4f}")
        c4.metric("Recall", f"{rec:.4f}")
        c5.metric("F1 Score", f"{f1:.4f}")
        c6.metric("MCC Score", f"{mcc:.4f}")
        
        st.markdown("---")
        col_a, col_b = st.columns(2)
        
        with col_a:
            st.subheader("Confusion Matrix")
            cm = confusion_matrix(y_test, y_pred)
            fig, ax = plt.subplots(figsize=(5, 4))
            ax.matshow(cm, cmap=plt.cm.Blues, alpha=0.3)
            for i in range(cm.shape[0]):
                for j in range(cm.shape[1]):
                    ax.text(x=j, y=i, s=cm[i, j], va='center', ha='center', size='xx-large')
            plt.xlabel("Predicted Label")
            plt.ylabel("True Label")
            st.pyplot(fig)
            
        with col_b:
            st.subheader("Classification Report")
            report = classification_report(y_test, y_pred, output_dict=True)
            st.dataframe(pd.DataFrame(report).transpose().style.format(precision=4), use_container_width=True)
else:
    st.info("Please upload your `test_data.csv` file using the sidebar panel to evaluate models.")
    st.markdown("### Preview of `test_data.csv` format:")
    st.dataframe(pd.read_csv('test_data.csv', nrows=5))
